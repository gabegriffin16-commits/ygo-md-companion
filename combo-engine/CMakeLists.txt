# combo-engine (AGPL-3.0-or-later). Build:
#   cmake -B build -DOCGCORE_DIR=<ygopro-core checkout with lua/src submodule> -DDEPS_DIR=<dir with json.hpp, miniz/, sqlite/>
#   cmake --build build --config Release
cmake_minimum_required(VERSION 3.16)
project(combo_engine C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(OCGCORE_DIR "" CACHE PATH "ygopro-core source (edo9300/ygopro-core)")
set(DEPS_DIR "" CACHE PATH "json.hpp, miniz/, sqlite/")
set(MIMALLOC_DIR "" CACHE PATH "microsoft/mimalloc source (optional, faster duel setup)")
if(MIMALLOC_DIR)
  add_library(mimalloc_static STATIC ${MIMALLOC_DIR}/src/static.c)
  target_include_directories(mimalloc_static PUBLIC ${MIMALLOC_DIR}/include)
  target_compile_definitions(mimalloc_static PRIVATE MI_STATIC_LIB)
endif()

# Lua 5.4 compiled as C++ with the engine's customization header in front, same as ygopro-core's meson build.
set(LUA_NAMES lapi lauxlib lbaselib lcode lctype ldebug ldo ldump lfunc lgc liolib llex lmathlib lmem lobject
    lopcodes lparser lstate lstring lstrlib ltable ltablib ltm lundump lvm lzio)
set(LUA_SRC)
foreach(n ${LUA_NAMES})
  set(w ${CMAKE_BINARY_DIR}/luawrap/${n}.cpp)
  set(pre "")
  if(MIMALLOC_DIR AND n STREQUAL "lauxlib")
    set(pre "#include \"${CMAKE_CURRENT_SOURCE_DIR}/src/mi_lua.h\"\n")
  endif()
  file(WRITE ${w}.tmp "${pre}#include \"${OCGCORE_DIR}/lua/luaconf-customize.h\"\n#include \"${OCGCORE_DIR}/lua/src/${n}.c\"\n")
  configure_file(${w}.tmp ${w} COPYONLY)
  list(APPEND LUA_SRC ${w})
endforeach()
add_library(lua STATIC ${LUA_SRC})
target_include_directories(lua PUBLIC ${OCGCORE_DIR}/lua/src)
if(MIMALLOC_DIR)
  target_link_libraries(lua PUBLIC mimalloc_static)
endif()

set(OCG_NAMES card duel effect field interpreter libcard libdebug libduel libeffect libgroup ocgapi operations
    playerop processor processor_visit scriptlib)
set(OCG_SRC)
foreach(n ${OCG_NAMES})
  list(APPEND OCG_SRC ${OCGCORE_DIR}/${n}.cpp)
endforeach()
add_library(ocgcore STATIC ${OCG_SRC})
target_include_directories(ocgcore PUBLIC ${OCGCORE_DIR})
target_link_libraries(ocgcore PUBLIC lua)

add_library(deps STATIC ${DEPS_DIR}/sqlite/sqlite3.c ${DEPS_DIR}/miniz/miniz.c)
target_include_directories(deps PUBLIC ${DEPS_DIR} ${DEPS_DIR}/sqlite ${DEPS_DIR}/miniz)
target_compile_definitions(deps PUBLIC SQLITE_OMIT_LOAD_EXTENSION SQLITE_THREADSAFE=1)

find_package(Threads REQUIRED)
add_executable(combo-engine src/main.cpp src/reset.cpp)
# Link-time optimization: most search time is spent inside the rules core and Lua (replaying lines), across library
# boundaries the compiler can only optimize when linking. ENGINE_LTO=OFF to turn it off.
option(ENGINE_LTO "Link-time optimization" ON)
if(ENGINE_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_ok OUTPUT ipo_msg)
  if(ipo_ok)
    set_property(TARGET combo-engine lua ocgcore deps PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  endif()
endif()
target_link_libraries(combo-engine PRIVATE ocgcore deps Threads::Threads)
if(MIMALLOC_DIR)
  target_compile_definitions(combo-engine PRIVATE MDC_MIMALLOC)
endif()
if(MSVC)
  set_property(TARGET combo-engine lua ocgcore deps PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(TARGET mimalloc_static)   # same static C runtime as the rest (mixing runtimes is fragile)
    set_property(TARGET mimalloc_static PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  target_compile_options(combo-engine PRIVATE /bigobj /utf-8)
elseif(MINGW)
  target_link_options(combo-engine PRIVATE -static)
endif()
